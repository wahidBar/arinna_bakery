<?php
/*
|--------------------------------------------------------------------------
| CATATAN SETUP — bootstrap/app.php (Laravel 11)
|--------------------------------------------------------------------------
| Laravel 11 tidak lagi punya app/Http/Kernel.php. Middleware alias dan
| route tambahan didaftarkan langsung di bootstrap/app.php. Tempelkan
| potongan berikut ke file bootstrap/app.php project Anda:
*/

use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Exceptions;
use Illuminate\Foundation\Configuration\Middleware;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        web: __DIR__ . '/../routes/web.php',
        commands: __DIR__ . '/../routes/console.php',
        health: '/up',
        then: function () {
            // Daftarkan routes/admin.php dengan prefix & middleware di sini
            \Illuminate\Support\Facades\Route::prefix('admin')
                ->name('admin.')
                ->middleware(['auth', 'verified', 'role:admin'])
                ->group(base_path('routes/admin.php'));
        },
    )
    ->withMiddleware(function (Middleware $middleware) {
        $middleware->alias([
            'role' => \App\Http\Middleware\EnsureUserIsAdmin::class,
        ]);
    })
    ->withExceptions(function (Exceptions $exceptions) {
        //
    })->create();
